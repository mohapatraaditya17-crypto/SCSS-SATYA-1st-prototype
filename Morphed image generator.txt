import random, csv, shutil
from pathlib import Path
import numpy as np, cv2
from PIL import Image

SRC = Path("source_images")
OUT = Path("test_images/tampered")
CTRL = Path("test_images/authentic")
N_TAMPERED = 20      # number of morphed images (change later to 5000)
N_AUTHENTIC = 5      # untouched control copies
MAXSIDE = 1600
random.seed(42)
OUT.mkdir(parents=True, exist_ok=True)
CTRL.mkdir(parents=True, exist_ok=True)

exts = {".jpg", ".jpeg", ".png", ".webp"}
sources = sorted(p for p in SRC.rglob("*") if p.suffix.lower() in exts)
print("Source images:", len(sources))
assert sources, "Put photos in the source_images folder first"

def load(p):
    im = Image.open(p).convert("RGB")
    im.thumbnail((MAXSIDE, MAXSIDE))
    return np.array(im)

def box(h, w, lo, hi):
    bw = max(32, int(w * random.uniform(lo, hi)))
    bh = max(32, int(h * random.uniform(lo, hi)))
    bw, bh = min(bw, w - 1), min(bh, h - 1)
    x, y = random.randint(0, w - bw), random.randint(0, h - bh)
    return x, y, bw, bh

def mask(bh, bw, soft):
    m = np.ones((bh, bw), np.float32)
    if soft:
        pad = max(2, int(min(bh, bw) * 0.12))
        m[:] = 0
        m[pad:bh - pad, pad:bw - pad] = 1
        m = cv2.GaussianBlur(m, (0, 0), pad / 2)
    return m[..., None]

def paste(img, patch, x, y, soft):
    bh, bw = patch.shape[:2]
    m = mask(bh, bw, soft)
    reg = img[y:y + bh, x:x + bw].astype(np.float32)
    img[y:y + bh, x:x + bw] = (patch * m + reg * (1 - m)).astype(np.uint8)

def splice(img, other):
    h, w = img.shape[:2]
    oh, ow = other.shape[:2]
    _, _, bw, bh = box(h, w, 0.08, 0.3)
    bw, bh = min(bw, ow - 1), min(bh, oh - 1)
    ox, oy = random.randint(0, ow - bw), random.randint(0, oh - bh)
    patch = other[oy:oy + bh, ox:ox + bw].astype(np.float32)
    x, y = random.randint(0, w - bw), random.randint(0, h - bh)
    paste(img, patch, x, y, random.random() < 0.5)

def copy_move(img, other=None):
    h, w = img.shape[:2]
    x, y, bw, bh = box(h, w, 0.08, 0.25)
    patch = img[y:y + bh, x:x + bw].astype(np.float32).copy()
    if random.random() < 0.3:
        patch = patch[:, ::-1]
    x2, y2 = random.randint(0, w - bw), random.randint(0, h - bh)
    paste(img, np.ascontiguousarray(patch), x2, y2, random.random() < 0.5)

def remove(img, other=None):
    h, w = img.shape[:2]
    x, y, bw, bh = box(h, w, 0.05, 0.2)
    m = np.zeros((h, w), np.uint8)
    m[y:y + bh, x:x + bw] = 255
    img[:] = cv2.inpaint(img, m, 5, cv2.INPAINT_TELEA)

def retouch(img, other=None):
    h, w = img.shape[:2]
    x, y, bw, bh = box(h, w, 0.1, 0.3)
    reg = img[y:y + bh, x:x + bw]
    if random.random() < 0.5:
        reg[:] = cv2.GaussianBlur(reg, (0, 0), random.uniform(2, 6))
    else:
        reg[:] = np.clip(reg.astype(np.float32) * random.uniform(0.7, 1.3)
                         + random.uniform(-25, 25), 0, 255).astype(np.uint8)

def fake_text(img, other=None):
    h, w = img.shape[:2]
    x, y, bw, bh = box(h, w, 0.1, 0.3)
    txt = str(random.randint(1000, 99999))
    cv2.putText(img, txt, (x, y + bh // 2), cv2.FONT_HERSHEY_SIMPLEX,
                random.uniform(0.8, 2.0), (random.randint(0, 255),) * 3, 2, cv2.LINE_AA)

OPS = {"splice": splice, "copymove": copy_move, "remove": remove,
       "retouch": retouch, "text": fake_text}

with open("labels.csv", "w", newline="", encoding="utf-8") as f:
    w = csv.writer(f)
    w.writerow(["file", "label", "manipulation", "source", "jpeg_quality"])
    for i in range(N_TAMPERED):
        src = sources[i % len(sources)]
        img = load(src).copy()
        name = random.choice(list(OPS))
        other = load(random.choice(sources)) if name == "splice" else None
        OPS[name](img, other)
        q = random.randint(70, 95)
        fn = OUT / f"morph_{i:05d}_{name}.jpg"
        Image.fromarray(img).save(fn, quality=q)
        w.writerow([str(fn), "tampered", name, src.name, q])
        if (i + 1) % 250 == 0:
            print(f"{i+1}/{N_TAMPERED} morphs created")
    for j, src in enumerate(random.sample(sources, min(N_AUTHENTIC, len(sources)))):
        fn = CTRL / f"auth_{j:05d}{src.suffix.lower()}"
        shutil.copy2(src, fn)
        w.writerow([str(fn), "authentic", "none", src.name, ""])
print("Done. Images in test_images, answer key in labels.csv")
